cmake_minimum_required(VERSION 3.20)
project(edge_ai_sdk LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(Python COMPONENTS Interpreter Development.Module REQUIRED)
find_package(pybind11 CONFIG REQUIRED)
set(ORT_ROOT "" CACHE PATH "Extracted official ONNX Runtime C/C++ release")
find_path(ORT_INCLUDE onnxruntime_cxx_api.h HINTS "${ORT_ROOT}/include" REQUIRED)
find_library(ORT_LIBRARY NAMES onnxruntime HINTS "${ORT_ROOT}/lib" REQUIRED)
pybind11_add_module(_native bindings/session.cpp)
target_include_directories(_native PRIVATE "${ORT_INCLUDE}")
target_link_libraries(_native PRIVATE "${ORT_LIBRARY}")
set_target_properties(_native PROPERTIES INSTALL_RPATH "$ORIGIN")
install(TARGETS _native LIBRARY DESTINATION edge_ai_sdk RUNTIME DESTINATION edge_ai_sdk)
if(WIN32)
  install(FILES "${ORT_ROOT}/lib/onnxruntime.dll" DESTINATION edge_ai_sdk)
else()
  install(DIRECTORY "${ORT_ROOT}/lib/" DESTINATION edge_ai_sdk FILES_MATCHING PATTERN "libonnxruntime.so*")
  # NuGet ships libonnxruntime.so without the SONAME .so.1 alias.
  install(FILES "${ORT_LIBRARY}" DESTINATION edge_ai_sdk RENAME libonnxruntime.so.1)
endif()
